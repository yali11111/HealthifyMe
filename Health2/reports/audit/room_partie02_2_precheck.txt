package com.example.health2.domain.usecase

/**
 * UseCase permettant de calculer les besoins
 * énergétiques quotidiens d'un utilisateur.
 *
 * Formule utilisée : Mifflin-St Jeor.
 *
 * Les résultats sont des estimations et ne constituent
 * pas une prescription nutritionnelle.
 */
class CalculateCaloriesUseCase {

    /**
     * Sexe utilisé pour le calcul métabolique.
     */
    enum class Sex {
        MALE,
        FEMALE
    }

    /**
     * Résultat du calcul calorique.
     */
    data class CaloriesResult(
        val basalMetabolicRate: Double,
        val dailyCalories: Double,
        val activityFactor: Double
    )

    /**
     * Calcule le métabolisme basal et les besoins quotidiens.
     *
     * @param weightKg Poids en kilogrammes.
     * @param heightCm Taille en centimètres.
     * @param age Âge en années.
     * @param sex Sexe utilisé par la formule.
     * @param activityFactor Coefficient d'activité physique.
     */
    operator fun invoke(
        weightKg: Double,
        heightCm: Double,
        age: Int,
        sex: Sex,
        activityFactor: Double = 1.2
    ): CaloriesResult {

        // Vérification du poids
        require(weightKg.isFinite() && weightKg > 0.0) {
            "Le poids doit être positif."
        }

        // Vérification de la taille
        require(heightCm.isFinite() && heightCm > 0.0) {
            "La taille doit être positive."
        }

        // Vérification de l'âge
        require(age in 1..120) {
            "L'âge doit être compris entre 1 et 120 ans."
        }

        // Vérification du niveau d'activité
        require(
            activityFactor.isFinite() &&
            activityFactor in 1.0..2.5
        ) {
            "Coefficient d'activité invalide."
        }

        // Ajustement selon le sexe utilisé par la formule
        val sexAdjustment = when (sex) {
            Sex.MALE -> 5.0
            Sex.FEMALE -> -161.0
        }

        // Calcul du métabolisme basal
        val bmr =
            (10.0 * weightKg) +
            (6.25 * heightCm) -
            (5.0 * age) +
            sexAdjustment

        // Estimation des besoins énergétiques quotidiens
        val dailyCalories = bmr * activityFactor

        // Retour du résultat
        return CaloriesResult(
            basalMetabolicRate = bmr,
            dailyCalories = dailyCalories,
            activityFactor = activityFactor
        )
    }
}

package com.example.health2.domain.usecase

/**
 * UseCase permettant d'analyser la durée du sommeil.
 *
 * Toutes les durées sont exprimées en minutes.
 */
class CalculateSleepUseCase {

    /**
     * Résultat de l'analyse du sommeil.
     */
    data class SleepResult(
        val totalDuration: Int,
        val deepSleep: Int,
        val lightSleep: Int,
        val remSleep: Int,
        val awakeDuration: Int,
        val sleepEfficiency: Float,
        val deepSleepPercentage: Float,
        val remSleepPercentage: Float
    )

    /**
     * Analyse les différentes phases du sommeil.
     *
     * @param deepSleep Sommeil profond en minutes.
     * @param lightSleep Sommeil léger en minutes.
     * @param remSleep Sommeil paradoxal en minutes.
     * @param awakeDuration Temps d'éveil en minutes.
     */
    operator fun invoke(
        deepSleep: Int,
        lightSleep: Int,
        remSleep: Int,
        awakeDuration: Int = 0
    ): SleepResult {

        require(deepSleep >= 0) {
            "La durée du sommeil profond est invalide."
        }

        require(lightSleep >= 0) {
            "La durée du sommeil léger est invalide."
        }

        require(remSleep >= 0) {
            "La durée du sommeil paradoxal est invalide."
        }

        require(awakeDuration >= 0) {
            "La durée d'éveil est invalide."
        }

        // Durée totale du sommeil
        val totalSleep = deepSleep + lightSleep + remSleep

        // Durée totale passée au lit
        val totalNight = totalSleep + awakeDuration

        // Efficacité du sommeil
        val efficiency = if (totalNight > 0) {
            totalSleep.toFloat() / totalNight.toFloat() * 100f
        } else {
            0f
        }

        // Pourcentage de sommeil profond
        val deepPercentage = if (totalSleep > 0) {
            deepSleep.toFloat() / totalSleep.toFloat() * 100f
        } else {
            0f
        }

        // Pourcentage de sommeil paradoxal
        val remPercentage = if (totalSleep > 0) {
            remSleep.toFloat() / totalSleep.toFloat() * 100f
        } else {
            0f
        }

        // Retour du résultat
        return SleepResult(
            totalDuration = totalSleep,
            deepSleep = deepSleep,
            lightSleep = lightSleep,
            remSleep = remSleep,
            awakeDuration = awakeDuration,
            sleepEfficiency = efficiency,
            deepSleepPercentage = deepPercentage,
            remSleepPercentage = remPercentage
        )
    }
}

package com.example.health2.domain.usecase

import com.example.health2.domain.repository.UserStatsRepository

 * d'activité physique d'un utilisateur.

 * Les statistiques récupérées comprennent notamment
 * les données relatives au nombre de pas.

    private val repository: UserStatsRepository

package com.example.health2.domain.usecase

/**
 * UseCase permettant de suivre un objectif de prise de poids.
 *
 * Les poids sont exprimés en kilogrammes.
 */
class WeightGainUseCase {

    /**
     * Résultat du suivi de prise de poids.
     */
    data class WeightGainResult(
        val currentWeight: Float,
        val targetWeight: Float,
        val gainedWeight: Float,
        val remainingWeight: Float,
        val progressPercentage: Float,
        val goalReached: Boolean
    )

    /**
     * Calcule la progression vers un objectif de prise de poids.
     *
     * @param initialWeight Poids de départ.
     * @param currentWeight Poids actuel.
     * @param targetWeight Poids souhaité.
     */
    operator fun invoke(
        initialWeight: Float,
        currentWeight: Float,
        targetWeight: Float
    ): WeightGainResult {

        // Vérification du poids initial
        require(initialWeight.isFinite() && initialWeight > 0f) {
            "Le poids initial est invalide."
        }

        // Vérification du poids actuel
        require(currentWeight.isFinite() && currentWeight > 0f) {
            "Le poids actuel est invalide."
        }

        // Vérification du poids cible
        require(
            targetWeight.isFinite() &&
            targetWeight > initialWeight
        ) {
            "Le poids cible doit être supérieur au poids initial."
        }

        // Quantité totale de poids à gagner
        val totalToGain = targetWeight - initialWeight

        // Quantité de poids gagnée
        val gained = currentWeight - initialWeight

        // Pourcentage de progression
        val progress = (
            gained / totalToGain
        ).coerceIn(0f, 1f) * 100f

        // Construction du résultat
        return WeightGainResult(
            currentWeight = currentWeight,
            targetWeight = targetWeight,
            gainedWeight = gained,
            remainingWeight = (targetWeight - currentWeight)
                .coerceAtLeast(0f),
            progressPercentage = progress,
            goalReached = currentWeight >= targetWeight
        )
    }
}
